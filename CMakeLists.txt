cmake_minimum_required(VERSION 3.16)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(IDF_TARGET esp32c3 CACHE STRING "Supported chip")
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(espc3_integrated VERSION 1.1.2)
if(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "This board profile supports ESP32-C3 only")
endif()
foreach(component app core display network debug_probe device_app main)
    idf_component_get_property(library ${component} COMPONENT_LIB)
    set_target_properties(${library} PROPERTIES CXX_STANDARD 20 CXX_EXTENSIONS ON)
    target_compile_options(${library} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-std=gnu++20>")
endforeach()
