cmake_minimum_required(VERSION 3.16)
project(c3_host_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(C3_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(C3_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Werror)
add_compile_options(-UNDEBUG)
enable_testing()
add_library(c3_core ../components/core/src/core.cpp ../components/core/src/render.cpp
    ../components/core/src/games.cpp ../components/core/src/bonus_games.cpp ../components/core/src/pet.cpp
    ../components/core/src/pc_monitor.cpp)
target_include_directories(c3_core PUBLIC ../components/core/include)
add_executable(core_tests core_tests.cpp)
target_link_libraries(core_tests PRIVATE c3_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(game_tests game_tests.cpp)
target_link_libraries(game_tests PRIVATE c3_core)
add_test(NAME game_tests COMMAND game_tests)
add_executable(pet_tests pet_tests.cpp)
target_link_libraries(pet_tests PRIVATE c3_core)
add_test(NAME pet_tests COMMAND pet_tests)
add_executable(pc_monitor_tests pc_monitor_tests.cpp)
target_link_libraries(pc_monitor_tests PRIVATE c3_core)
add_test(NAME pc_monitor_tests COMMAND pc_monitor_tests)
add_executable(tetris_input_tests tetris_input_tests.cpp)
target_link_libraries(tetris_input_tests PRIVATE c3_core)
add_test(NAME tetris_input_tests COMMAND tetris_input_tests)
add_executable(lifecycle_tests lifecycle_tests.cpp ../components/app/src/app_manager.cpp)
target_include_directories(lifecycle_tests PRIVATE ../components/app/include)
add_test(NAME lifecycle_tests COMMAND lifecycle_tests)
add_executable(dap_protocol_tests dap_protocol_tests.cpp)
target_include_directories(dap_protocol_tests PRIVATE ../components/debug_probe/include)
add_test(NAME dap_protocol_tests COMMAND dap_protocol_tests)
add_executable(dap_engine_tests dap_engine_tests.cpp
    ../components/debug_probe/vendor/DAP.c ../components/debug_probe/vendor/SW_DP.c)
target_include_directories(dap_engine_tests PRIVATE dap_stubs ../components/debug_probe/include
    ../components/debug_probe/src ../components/debug_probe/vendor)
target_compile_options(dap_engine_tests PRIVATE -Wno-unused-variable)
add_test(NAME dap_engine_tests COMMAND dap_engine_tests)
add_executable(render_previews render_previews.cpp)
target_link_libraries(render_previews PRIVATE c3_core)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME dap_client_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/client_tests.py)
